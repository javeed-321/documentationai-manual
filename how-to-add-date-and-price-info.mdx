```curl
edgeWorkerId="42"

  --url "https://$hostname/edgeworkers/v1/ids/$edgeWorkerId" \
  --header "Accept: application/json"
```

```node
const EdgeGrid = require("akamai-edgegrid");
const os = require("os");
const edgercPath = os.homedir + "/.edgerc.txt";
const edgercSection = "default";  
const eg = new EdgeGrid({
  path: edgercPath,
  section: edgercSection
});   

const edgeWorkerId = "42";

eg.auth({
  path: `/edgeworkers/v1/ids/${edgeWorkerId}`,
  method: "GET",
  headers: {
    
    "Accept": "application/json"}                    
});
eg.send(function(error, response, body) {
  console.log(body);    
});
```

```python
import requests
from akamai.edgegrid import EdgeGridAuth, EdgeRc
from urllib.parse import urljoin

edgerc = EdgeRc('~/.edgerc.txt')
section = 'default'
baseurl = 'https://%s' % edgerc.get(section, 'host')

session = requests.Session()
session.auth = EdgeGridAuth.from_edgerc(edgerc, section)
edgeWorkerId = "42"
 

path= '/edgeworkers/v1/ids/{}'.format(edgeWorkerId)
headers = {
  
  "Accept": "application/json"}  

result = session.get(urljoin(baseurl, path), headers=headers)
print(result.status_code)
print(result.json()) 
```
